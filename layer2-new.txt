CREATE OR ALTER FUNCTION dbo.GetTemplateDB() RETURNS SYSNAME AS BEGIN
DECLARE @db SYSNAME;
SELECT TOP 1 @db=DBName FROM dbo.DiscoveryConfig WHERE IsActive=1 AND DatabaseRole=N'Template';
IF @db IS NULL SELECT TOP 1 @db=DBName FROM dbo.DiscoveryConfig WHERE IsActive=1 AND IsTemplate=1;
IF @db IS NULL SELECT TOP 1 @db=DBName FROM dbo.DiscoveryConfig WHERE IsActive=1 ORDER BY Ord;
RETURN @db;
END;
GO
CREATE OR ALTER FUNCTION dbo.GetMasterSourceDB() RETURNS SYSNAME AS BEGIN

RETURN @db;
END;

-- ═══ SINGLE CONSOLIDATED GATEKEEPER PROCEDURE ═══
CREATE OR ALTER PROCEDURE dbo.usp_GateKeeperPass @VId UNIQUEIDENTIFIER OUTPUT, @Passed BIT OUTPUT, @ErrCnt INT OUTPUT, @ShowSel BIT = 1
AS
BEGIN
    SET NOCOUNT ON;
    SET @VId = NEWID();
    SET @ErrCnt = 0;
    SET @Passed = 1;

    DECLARE @TplDB SYSNAME = (SELECT TOP 1 DBName FROM dbo.DiscoveryConfig WHERE IsActive = 1 AND IsTemplate = 1);
    DECLARE @MstrDB SYSNAME = (SELECT TOP 1 DBName FROM dbo.DiscoveryConfig WHERE IsActive = 1 AND DatabaseRole = N'HO');
    IF OBJECT_ID('dbo.ContractValidationResults', 'U') IS NOT NULL DROP TABLE dbo.ContractValidationResults;
    CREATE TABLE dbo.ContractValidationResults(CheckId INT IDENTITY PRIMARY KEY,Category NVARCHAR(32) NOT NULL,Severity NVARCHAR(16) NOT NULL,SchemaName SYSNAME NULL,TableName SYSNAME NULL,Message NVARCHAR(512) NOT NULL,CreatedAt DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME());

    IF @MstrDB IS NULL
    BEGIN
        INSERT INTO dbo.ContractValidationResults(Category, Severity, Message)
        VALUES (N'Config', N'ERROR', N'No HO database configured.');
        SET @ErrCnt = 1;
        SET @Passed = 0;
        RETURN;
    END

    -- CHECK 1: Manifest Consistency
    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Manifest', N'ERROR', k.SchemaName, k.TableName, N'KSP has no MigrationExecutionContract.'
    FROM dbo.KeyStrategyPlan k
    WHERE NOT EXISTS(SELECT 1 FROM dbo.MigrationExecutionContract c
                     WHERE c.SchemaName = k.SchemaName AND c.TableName = k.TableName);

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Contract', N'ERROR', SchemaName, TableName, N'Required fields NULL.'
    FROM dbo.MigrationExecutionContract
    WHERE UniquenessStrategy IS NULL OR MigrationMethod IS NULL OR ExecutionPhaseId IS NULL;

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Validation', N'ERROR', c.SchemaName, c.TableName, N'Method ' + c.MigrationMethod + N' needs validation.'
    FROM dbo.MigrationExecutionContract c
    WHERE c.IsUnsupported = 0
      AND c.GovernanceStatus NOT IN(N'PENDING_REVIEW', N'REJECTED')
      AND c.MigrationMethod IN(N'INSERT_ONLY', N'REKEY', N'EXPAND_PK', N'MERGE_TO_MASTER', N'MERGE_LOOKUP', N'OVERWRITE', N'PRESERVE')
      AND NOT EXISTS(SELECT 1 FROM dbo.ValidationPlan v
                     WHERE v.SchemaName = c.SchemaName AND v.TableName = c.TableName);

    -- CHECK 2: Topology
    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'FKRewrite', N'ERROR', k.SchemaName, k.TableName, N'ExpandPK non-self FKs no FKRewritePlan.'
    FROM dbo.KeyStrategyPlan k
    WHERE k.UniquenessStrategy = N'ExpandPK'
      AND k.IsUnsupported = 0
      AND k.GovernanceStatus NOT IN(N'PENDING_REVIEW', N'REJECTED')
      AND NOT EXISTS(SELECT 1 FROM dbo.FKRewritePlan f
                     WHERE f.ParentSchema = k.SchemaName AND f.ParentTable = k.TableName)
      AND EXISTS(SELECT 1 FROM dbo.TableFKDetails fk
                 WHERE fk.DatabaseName = @TplDB AND fk.ReferencedSchema = k.SchemaName
                   AND fk.ReferencedTable = k.TableName AND fk.IsSelfReference = 0);

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Topology', N'ERROR', c1.SchemaName, c1.TableName,
           N'Child ExecOrder (' + CAST(c1.ExecutionOrder AS NVARCHAR) + N') <= parent '
           + fk.ReferencedTable + N' (' + CAST(c2.ExecutionOrder AS NVARCHAR) + N').'
    FROM dbo.MigrationExecutionContract c1
    JOIN dbo.TableFKDetails fk ON fk.SchemaName = c1.SchemaName AND fk.TableName = c1.TableName
        AND fk.DatabaseName = @TplDB AND fk.IsSelfReference = 0
    JOIN dbo.MigrationExecutionContract c2 ON c2.SchemaName = fk.ReferencedSchema AND c2.TableName = fk.ReferencedTable
    WHERE c1.ExecutionOrder <= c2.ExecutionOrder
      AND c1.IsUnsupported = 0 AND c2.IsUnsupported = 0;

    -- CHECK 3: Source Plan
    DECLARE @RegionCnt INT = (SELECT COUNT(*) FROM dbo.DiscoveryConfig WHERE IsActive = 1 AND DatabaseRole = N'Region');
    DECLARE @Exp INT, @Act INT;

    SELECT @Exp = SUM(CASE
        WHEN SourcePolicy = N'HO_ONLY'     THEN 1 + @RegionCnt
        WHEN SourcePolicy = N'REGION_ONLY' THEN @RegionCnt
        WHEN SourcePolicy = N'BOTH'        THEN 1 + @RegionCnt
        ELSE 0 END)
    FROM dbo.MigrationExecutionContract
    WHERE IsUnsupported = 0 AND GovernanceStatus NOT IN(N'REJECTED', N'PENDING_REVIEW');

    SET @Act = (SELECT COUNT(*) FROM dbo.MigrationSourcePlan);

    IF @Act <> @Exp
        INSERT INTO dbo.ContractValidationResults(Category, Severity, Message)
        VALUES (N'SrcPlan', N'ERROR', N'Expected ' + CAST(@Exp AS NVARCHAR) + N' found ' + CAST(@Act AS NVARCHAR));

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Missing', N'ERROR', c.SchemaName, c.TableName, N'No source plan for ' + dc.DBName
    FROM dbo.MigrationExecutionContract c
    CROSS JOIN dbo.DiscoveryConfig dc
    WHERE c.IsUnsupported = 0
      AND c.GovernanceStatus NOT IN(N'PENDING_REVIEW', N'REJECTED')
      AND dc.IsActive = 1
      AND ((c.SourcePolicy = N'HO_ONLY'     AND dc.DatabaseRole IN(N'HO', N'Region'))
        OR (c.SourcePolicy = N'REGION_ONLY' AND dc.DatabaseRole = N'Region')
        OR (c.SourcePolicy = N'BOTH'        AND dc.DatabaseRole IN(N'HO', N'Region')))
      AND NOT EXISTS(SELECT 1 FROM dbo.MigrationSourcePlan m
                     WHERE m.ContractId = c.ContractId AND m.SourceDB = dc.DBName);

    -- CHECK 4: Master References
    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'MTM_Consistency', N'ERROR', k.SchemaName, k.TableName, N'MergeToMaster MasterKey/Entity NULL.'
    FROM dbo.KeyStrategyPlan k
    WHERE k.MigrationStrategy = N'MergeToMaster'
      AND (k.MasterKeyColumn IS NULL OR k.MasterEntity = N'Unknown');

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'HO_Required', N'ERROR', k.SchemaName, k.TableName, N'MTM HO empty, region has data.'
    FROM dbo.KeyStrategyPlan k
    WHERE k.MigrationStrategy = N'MergeToMaster'
      AND k.IsUnsupported = 0
      AND NOT EXISTS(SELECT 1 FROM dbo.TableDetails td
                     WHERE td.DatabaseName = @MstrDB AND td.SchemaName = k.SchemaName
                       AND td.TableName = k.TableName AND td.[RowCount] > 0)
      AND EXISTS(SELECT 1 FROM dbo.TableDetails td2
                 JOIN dbo.DiscoveryConfig dc ON dc.DBName = td2.DatabaseName AND dc.DatabaseRole = N'Region' AND dc.IsActive = 1
                 WHERE td2.SchemaName = k.SchemaName AND td2.TableName = k.TableName AND td2.[RowCount] > 0);

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'HO_ONLY_REKEY', N'ERROR', c.SchemaName, c.TableName, N'HO_ONLY + REKEY contradiction.'
    FROM dbo.MigrationExecutionContract c
    WHERE c.SourcePolicy = N'HO_ONLY' AND c.MigrationMethod = N'REKEY' AND c.IsUnsupported = 0;

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Regression', N'WARNING', ec.SchemaName, ec.TableName,
           N'Mismatch: ' + STUFF(
               (CASE WHEN ec.ExpectedCategory IS NOT NULL AND (ma.WinnerCategory <> ec.ExpectedCategory OR ma.WinnerCategory IS NULL)
                     THEN N'; Cat expected ' + ec.ExpectedCategory + N' got ' + ISNULL(ma.WinnerCategory, N'?') ELSE N'' END)
             + (CASE WHEN ec.ExpectedStrategy IS NOT NULL AND (ma.MigrationStrategy <> ec.ExpectedStrategy OR ma.MigrationStrategy IS NULL)
                     THEN N'; Strat expected ' + ec.ExpectedStrategy + N' got ' + ISNULL(ma.MigrationStrategy, N'?') ELSE N'' END)
             + (CASE WHEN ec.ExpectedSourcePolicy IS NOT NULL AND (ma.SourcePolicy <> ec.ExpectedSourcePolicy OR ma.SourcePolicy IS NULL)
                     THEN N'; Pol expected ' + ec.ExpectedSourcePolicy + N' got ' + ISNULL(ma.SourcePolicy, N'?') ELSE N'' END), 1, 2, N'')
    FROM dbo.ExpectedClassification ec
    LEFT JOIN dbo.MigrationAnalysis ma ON ma.SchemaName = ec.SchemaName AND ma.TableName = ec.TableName
    WHERE (ec.ExpectedCategory IS NOT NULL AND (ma.WinnerCategory <> ec.ExpectedCategory OR ma.WinnerCategory IS NULL))
       OR (ec.ExpectedStrategy IS NOT NULL AND (ma.MigrationStrategy <> ec.ExpectedStrategy OR ma.MigrationStrategy IS NULL))
       OR (ec.ExpectedSourcePolicy IS NOT NULL AND (ma.SourcePolicy <> ec.ExpectedSourcePolicy OR ma.SourcePolicy IS NULL));

    -- CHECK 5: Schema Drift
    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Drift', N'ERROR', SchemaName, TableName, N'Blocking drift ' + DriftType + N' on ' + ColumnName
    FROM dbo.SchemaDriftMatrix
    WHERE DriftSeverity = N'ERROR' AND DriftType NOT IN(N'MISSING_COLUMN', N'TYPE_MISMATCH');

    -- CHECK 6: Plan Consistency
    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'PlanOrphan', N'ERROR', cp.SchemaName, cp.TableName, N'CategoryPlan no StrategyPlan.'
    FROM dbo.CategoryPlan cp
    WHERE NOT EXISTS(SELECT 1 FROM dbo.StrategyPlan sp
                     WHERE sp.SchemaName = cp.SchemaName AND sp.TableName = cp.TableName);

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'PlanOrphan', N'ERROR', sp.SchemaName, sp.TableName, N'StrategyPlan no SourcePolicyPlan.'
    FROM dbo.StrategyPlan sp
    WHERE NOT EXISTS(SELECT 1 FROM dbo.SourcePolicyPlan spp
                     WHERE spp.SchemaName = sp.SchemaName AND spp.TableName = sp.TableName);

    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'SPP_Contradiction', N'ERROR', spp.SchemaName, spp.TableName, N'HO_ONLY + Rekey.'
    FROM dbo.SourcePolicyPlan spp
    JOIN dbo.StrategyPlan sp ON sp.SchemaName = spp.SchemaName AND sp.TableName = spp.TableName
    WHERE spp.SourcePolicy = N'HO_ONLY' AND sp.MigrationStrategy = N'Rekey';

    -- CHECK 7: Unsupported
    INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
    SELECT N'Unsupported',
           CASE WHEN UnsupportedReason LIKE N'No data source:%' THEN N'INFO' ELSE N'WARNING' END,
           SchemaName, TableName, N'Unsupported: ' + ISNULL(UnsupportedReason, N'<none>')
    FROM dbo.MigrationExecutionContract WHERE IsUnsupported = 1;

    -- CHECK 8: Manual Overrides
    IF OBJECT_ID('dbo.ManualClassificationOverride', 'U') IS NOT NULL
        INSERT INTO dbo.ContractValidationResults(Category, Severity, SchemaName, TableName, Message)
        SELECT N'ManualOverride', N'INFO', mco.SchemaName, mco.TableName,
               N'Override: ' + ISNULL(mco.ForcedStrategy, N'<none>') + N'/' + ISNULL(mco.ForcedMigrationStrategy, N'<none>') + N'. ' + mco.Reason
        FROM dbo.ManualClassificationOverride mco;

    -- Calculate final status
    SELECT @ErrCnt = COUNT(*) FROM dbo.ContractValidationResults WHERE Severity = N'ERROR';
    SET @Passed = CASE WHEN @ErrCnt > 0 THEN 0 ELSE 1 END;

    IF @ErrCnt > 0
        PRINT N'[GATEKEEPER] FAILED with ' + CAST(@ErrCnt AS NVARCHAR) + N' error(s).';
    ELSE
        PRINT N'[GATEKEEPER] PASSED.';

    IF @ShowSel = 1
    BEGIN
        SELECT @TplDB AS TplDB, @MstrDB AS MstrDB, @Passed AS Passed,
               @ErrCnt AS Errors,
               (SELECT COUNT(*) FROM dbo.ContractValidationResults WHERE Severity = N'WARNING') AS Warnings;

        SELECT Category, Severity, SchemaName, TableName, Message
        FROM dbo.ContractValidationResults
        ORDER BY CASE Severity WHEN N'ERROR' THEN 1 WHEN N'WARNING' THEN 2 ELSE 3 END, Category;
    END
END;
GO

DECLARE @TemplateDB SYSNAME=dbo.GetTemplateDB();
WITH Agg AS(SELECT td.SchemaName,td.TableName,COUNT(DISTINCT td.DatabaseName)AS DBCount,COUNT(DISTINCT td.[RowCount])AS DistinctRC,MAX(td.[RowCount])AS MaxRC,AVG(CAST(td.[RowCount]AS DECIMAL(18,2)))AS AvgRC,STDEV(CAST(td.[RowCount]AS DECIMAL(18,2)))AS StdRC,MIN(ts.MinPK)AS GMinPK,MAX(ts.MinPK)AS GMaxMinPK,MIN(ts.MaxPK)AS GMinMaxPK,MAX(ts.MaxPK)AS GMaxPK FROM dbo.TableDetails td LEFT JOIN dbo.TableStatistics ts ON ts.DatabaseName=td.DatabaseName AND ts.SchemaName=td.SchemaName AND ts.TableName=td.TableName GROUP BY td.SchemaName,td.TableName)
INSERT INTO dbo.CrossDatabaseComparison SELECT SchemaName,TableName,CASE WHEN DistinctRC=1 THEN 1 ELSE 0 END,CASE WHEN AvgRC>0 THEN CAST(ROUND(StdRC*100.0/AvgRC,2)AS DECIMAL(10,2)) ELSE 0 END,CASE WHEN GMaxMinPK<=GMinMaxPK AND DBCount>1 THEN 1 ELSE 0 END,MaxRC,GMinPK,GMaxPK,DBCount FROM Agg;

INSERT INTO dbo.DependencyGraph(SchemaName,TableName,ParentCount,ChildCount,FKDepthStatus)
SELECT DISTINCT SchemaName,TableName,
(SELECT COUNT(*)FROM dbo.TableFKDetails fk WHERE fk.DatabaseName=@TemplateDB AND fk.SchemaName=td.SchemaName AND fk.TableName=td.TableName AND fk.IsSelfReference=0),
(SELECT COUNT(DISTINCT fk2.TableName)FROM dbo.TableFKDetails fk2 WHERE fk2.DatabaseName=@TemplateDB AND fk2.ReferencedSchema=td.SchemaName AND fk2.ReferencedTable=td.TableName AND fk2.IsSelfReference=0),
CASE WHEN (SELECT COUNT(*)FROM dbo.TableFKDetails fk WHERE fk.DatabaseName=@TemplateDB AND fk.SchemaName=td.SchemaName AND fk.TableName=td.TableName AND fk.IsSelfReference=0)=0 THEN N'ROOT' ELSE N'PENDING' END
FROM dbo.TableDetails td WHERE td.DatabaseName=@TemplateDB;

DECLARE @d INT=0,@u INT=1;
UPDATE dg SET FKDepth=0,FKDepthStatus=N'ROOT' FROM dbo.DependencyGraph dg WHERE ParentCount=0;
WHILE @u>0 BEGIN
;WITH Ready AS(
    SELECT dg.SchemaName,dg.TableName,MAX(p.FKDepth)+1 AS NewDepth
    FROM dbo.DependencyGraph dg
    JOIN dbo.TableFKDetails fk ON fk.DatabaseName=@TemplateDB AND fk.SchemaName=dg.SchemaName AND fk.TableName=dg.TableName AND fk.IsSelfReference=0
    JOIN dbo.DependencyGraph p ON p.SchemaName=fk.ReferencedSchema AND p.TableName=fk.ReferencedTable
    WHERE dg.FKDepth IS NULL
    GROUP BY dg.SchemaName,dg.TableName
    HAVING COUNT(*)=SUM(CASE WHEN p.FKDepth IS NOT NULL THEN 1 ELSE 0 END)
)
UPDATE dg SET dg.FKDepth=r.NewDepth,dg.FKDepthStatus=N'RESOLVED'
FROM dbo.DependencyGraph dg JOIN Ready r ON r.SchemaName=dg.SchemaName AND r.TableName=dg.TableName;
SET @u=@@ROWCOUNT; SET @d+=1;
IF @d>1000 BREAK;  -- safety valve against infinite loop on pathological graphs
END
UPDATE dg SET FKDepth=900,FKDepthStatus=N'CYCLE',NeedsReview=1 FROM dbo.DependencyGraph dg WHERE FKDepth IS NULL AND ParentCount>0;
UPDATE dbo.DependencyGraph SET IsHub=CASE WHEN ChildCount>=10 THEN 1 ELSE 0 END,IsLeaf=CASE WHEN ParentCount>0 AND ChildCount=0 THEN 1 ELSE 0 END;
UPDATE dg SET dg.RoughCategory=CASE WHEN dg.ChildCount>=5 AND dg.ParentCount=0 THEN N'Lookup' WHEN dg.ChildCount>=3 AND dg.ParentCount>=1 THEN N'Master' WHEN dg.ParentCount>=3 AND dg.ChildCount>=1 THEN N'TransactionHeader' WHEN dg.ParentCount>=1 AND dg.ChildCount=0 THEN N'TransactionDetail' ELSE N'Unknown' END FROM dbo.DependencyGraph dg;
UPDATE dbo.DependencyGraph SET TxnChainDepth=0 WHERE RoughCategory IN(N'Lookup',N'Master');
DECLARE @cd INT=0,@cu INT=1;
WHILE @cu>0 BEGIN UPDATE dg SET dg.TxnChainDepth=@cd+1 FROM dbo.DependencyGraph dg WHERE dg.TxnChainDepth IS NULL AND dg.RoughCategory IN(N'TransactionHeader',N'TransactionDetail') AND EXISTS(SELECT 1 FROM dbo.TableFKDetails fk JOIN dbo.DependencyGraph p ON p.SchemaName=fk.ReferencedSchema AND p.TableName=fk.ReferencedTable WHERE fk.DatabaseName=@TemplateDB AND fk.SchemaName=dg.SchemaName AND fk.TableName=dg.TableName AND fk.IsSelfReference=0 AND p.TxnChainDepth=@cd); SET @cu=@@ROWCOUNT; SET @cd+=1; END
UPDATE dbo.DependencyGraph SET TxnChainDepth=99 WHERE TxnChainDepth IS NULL;
-- FIX: cycle/unresolved rows already got FKDepth=900 above; anything still NULL here has ParentCount=0 miscounted or is truly isolated -- treat as CYCLE (900), NOT as ROOT (0), to avoid masking topology violations
UPDATE dbo.DependencyGraph SET FKDepth=900,FKDepthStatus=N'UNKNOWN',NeedsReview=1 WHERE FKDepth IS NULL;

INSERT INTO dbo.NaturalKeyCandidates(SchemaName,TableName,IndexName,KeyColumns) SELECT SchemaName,TableName,IndexName,KeyColumns FROM dbo.TableIndexes WHERE IsUnique=1 AND IsPrimaryKey=0 AND DatabaseName=@TemplateDB;
WITH Ranked AS(SELECT SchemaName,TableName,IndexName,ROW_NUMBER()OVER(PARTITION BY SchemaName,TableName ORDER BY LEN(KeyColumns)-LEN(REPLACE(KeyColumns,N',',N'')))AS rnk FROM dbo.NaturalKeyCandidates) UPDATE nk SET nk.CandidateRank=r.rnk FROM dbo.NaturalKeyCandidates nk JOIN Ranked r ON r.SchemaName=nk.SchemaName AND r.TableName=nk.TableName AND r.IndexName=nk.IndexName;

-- Populate IdentityPlan from discovered metadata
INSERT INTO dbo.IdentityPlan(SchemaName, TableName, HOColumnCount, RegionColumnCount, ExistsInHO, ExistsInRegion, HasBranchIdInPK, IsCompositePK, HasIdentityColumn, TotalIncomingRefs)
SELECT
    td.SchemaName,
    td.TableName,
    -- HOColumnCount: Count explicit HO ownership columns
    (SELECT COUNT(*) FROM dbo.TableColumns tc
     WHERE tc.DatabaseName = @TemplateDB AND tc.SchemaName = td.SchemaName AND tc.TableName = td.TableName
     AND (tc.ColumnName LIKE N'HO%' OR tc.ColumnName LIKE N'%HOID' OR tc.ColumnName LIKE N'%HO_ID')),
    -- RegionColumnCount: Count explicit Region ownership columns
    (SELECT COUNT(*) FROM dbo.TableColumns tc
     WHERE tc.DatabaseName = @TemplateDB AND tc.SchemaName = td.SchemaName AND tc.TableName = td.TableName
     AND (tc.ColumnName LIKE N'BranchId%' OR tc.ColumnName LIKE N'RegionId%')),
    -- ExistsInHO
    CASE WHEN EXISTS(SELECT 1 FROM dbo.TableDetails td2 WHERE td2.DatabaseName = @MasterSourceDB AND td2.SchemaName = td.SchemaName AND td2.TableName = td.TableName) THEN 1 ELSE 0 END,
    -- ExistsInRegion
    CASE WHEN EXISTS(SELECT 1 FROM dbo.TableDetails td3 JOIN dbo.DiscoveryConfig dc ON td3.DatabaseName = dc.DBName WHERE td3.DatabaseName NOT IN (@MasterSourceDB, @TemplateDB) AND dc.DatabaseRole = N'Region' AND td3.SchemaName = td.SchemaName AND td3.TableName = td.TableName) THEN 1 ELSE 0 END,
    -- HasBranchIdInPK
    CASE WHEN EXISTS(SELECT 1 FROM dbo.TablePKColumns pk WHERE pk.DatabaseName = @TemplateDB AND pk.SchemaName = td.SchemaName AND pk.TableName = td.TableName AND pk.ColumnName IN(N'BranchId',N'BranchID',N'RegionId',N'RegionID')) THEN 1 ELSE 0 END,
    -- IsCompositePK
    CASE WHEN (SELECT COUNT(*) FROM dbo.TablePKColumns pk WHERE pk.DatabaseName = @TemplateDB AND pk.SchemaName = td.SchemaName AND pk.TableName = td.TableName) > 1 THEN 1 ELSE 0 END,
    -- HasIdentityColumn
    CASE WHEN EXISTS(SELECT 1 FROM dbo.TableColumns tc WHERE tc.DatabaseName = @TemplateDB AND tc.SchemaName = td.SchemaName AND tc.TableName = td.TableName AND tc.IsIdentity = 1) THEN 1 ELSE 0 END,
    -- TotalIncomingRefs
    ISNULL((SELECT COUNT(*) FROM dbo.TableFKDetails fk WHERE fk.ReferencedSchema = td.SchemaName AND fk.ReferencedTable = td.TableName AND fk.DatabaseName = @TemplateDB), 0)
FROM dbo.TableDetails td
WHERE td.DatabaseName = @TemplateDB;

-- Update HasMasterKey and MasterKeyColumn
UPDATE ip SET
    HasMasterKey = 1,
    MasterKeyColumn = mc.MasterKeyColumn,
    IdentityOwnerColumn = mc.MasterKeyColumn
FROM dbo.IdentityPlan ip
CROSS APPLY (
    SELECT TOP 1 tc.ColumnName AS MasterKeyColumn
    FROM dbo.TableColumns tc
    WHERE tc.DatabaseName = @TemplateDB AND tc.SchemaName = ip.SchemaName AND tc.TableName = ip.TableName
    AND (tc.ColumnName LIKE N'%MasterKey%' OR tc.ColumnName LIKE N'%BusinessKey%' OR tc.ColumnName LIKE N'%NaturalKey%')
    ORDER BY tc.ColumnName
) mc;

UPDATE ip SET
    IdentityParent = parent.ParentName
FROM dbo.IdentityPlan ip
CROSS APPLY (
    SELECT TOP 1 fk.ReferencedSchema + N'.' + fk.ReferencedTable AS ParentName
    FROM dbo.TableFKDetails fk
    WHERE fk.DatabaseName=@TemplateDB
      AND fk.SchemaName=ip.SchemaName
      AND fk.TableName=ip.TableName
      AND ISNULL(fk.IsSelfReference,0)=0
    ORDER BY fk.ReferencedSchema,fk.ReferencedTable
) parent;

-- Initial IdentityOwner assignment based on explicit signals
UPDATE dbo.IdentityPlan SET
    IdentityOwner = CASE
        WHEN HOColumnCount > 0 THEN N'HO'
        WHEN RegionColumnCount > 0 THEN N'Region'
        WHEN IsCompositePK = 1 AND HasBranchIdInPK = 1 THEN N'Composite'
        WHEN HasIdentityColumn = 1 AND ExistsInHO = 0 AND ExistsInRegion = 1 THEN N'Generated'
        WHEN ExistsInHO = 1 AND ExistsInRegion = 0 THEN N'HO'
        WHEN ExistsInHO = 0 AND ExistsInRegion = 1 THEN N'Region'
        ELSE N'Unknown'
    END,
    IdentityResolution = CASE
        WHEN HOColumnCount > 0 THEN N'Preserve'
        WHEN RegionColumnCount > 0 THEN N'Preserve'
        WHEN IsCompositePK = 1 THEN N'Preserve'
        WHEN HasIdentityColumn = 1 THEN N'Generate'
        ELSE N'Unknown'
    END,
    OwnershipRule = N'PASS1_EXPLICIT_SIGNALS',
    ReasonCode = CASE
        WHEN HOColumnCount > 0 THEN N'HO_COLUMNS'
        WHEN RegionColumnCount > 0 THEN N'REGION_COLUMNS'
        WHEN IsCompositePK = 1 THEN N'COMPOSITE_PK'
        WHEN HasIdentityColumn = 1 THEN N'IDENTITY_COLUMN'
        WHEN ExistsInHO = 1 AND ExistsInRegion = 0 THEN N'HO_ONLY'
        WHEN ExistsInHO = 0 AND ExistsInRegion = 1 THEN N'REGION_ONLY'
        ELSE N'UNKNOWN'
    END;

DECLARE @ipCnt INT = @@ROWCOUNT;
PRINT N'[PHASE 2.ID] Pass 1: ' + CAST(@ipCnt AS NVARCHAR) + N' tables.';

-- ═══ PHASE 2.ID.SCORE: AGGREGATE EVIDENCE WITH TWO-PASS RANKING ═══════════════════════════════
PRINT N'[PHASE 2.ID.SCORE] Aggregating evidence scores...';

-- EvidenceScore table stores final aggregated scores per candidate
IF OBJECT_ID('dbo.EvidenceScore','U') IS NOT NULL DROP TABLE dbo.EvidenceScore;
CREATE TABLE dbo.EvidenceScore(
    SchemaName SYSNAME NOT NULL,
    TableName SYSNAME NOT NULL,
    OwnerCandidate NVARCHAR(32) NOT NULL, -- HO, Region, Composite, Shared, Generated
    NetScore INT NOT NULL DEFAULT 0,
    SignalCount INT NOT NULL DEFAULT 0,
    DistinctSignalTypes INT NOT NULL DEFAULT 0,
    Signals NVARCHAR(MAX) NULL,
    RankInTable INT NULL,
    WinningMargin INT NULL,
    Confidence DECIMAL(5,2) NULL,
    CONSTRAINT PK_ES PRIMARY KEY(SchemaName, TableName, OwnerCandidate)
);

-- Two-pass aggregation for accurate runner-up calculation
WITH EffectiveWeights AS (
    -- Calculate effective weight = BaseWeight × (RawConfidence / 100)
    SELECT
        oe.SchemaName,
        oe.TableName,
        oe.SignalName,
        oe.SignalValue,
        ewc.BaseWeight,
        ewc.ContributionType,
        ewc.AppliesToCandidate,
        oe.RawConfidence,
        CAST(ewc.BaseWeight * (oe.RawConfidence / 100.0) AS INT) AS EffectiveWeight
    FROM dbo.OwnershipEvidence oe
    JOIN dbo.EvidenceWeightConfig ewc ON oe.SignalName = ewc.SignalName
    WHERE ewc.IsActive = 1
),
Aggregated AS (
    -- First pass: aggregate scores per candidate
    SELECT
        ew.SchemaName,
        ew.TableName,
        ew.AppliesToCandidate AS OwnerCandidate,
        SUM(CASE WHEN ew.ContributionType = N'Support' THEN ew.EffectiveWeight
                 WHEN ew.ContributionType = N'Contradict' THEN -ew.EffectiveWeight
                 ELSE 0 END) AS NetScore,
        COUNT(*) AS SignalCount,
        COUNT(DISTINCT ew.SignalName) AS DistinctSignalTypes,
        STRING_AGG(ew.SignalName + N'(' + CAST(ew.EffectiveWeight AS NVARCHAR) + N')', N', ') AS Signals
    FROM EffectiveWeights ew
    GROUP BY ew.SchemaName, ew.TableName, ew.AppliesToCandidate
),
Ranked AS (
    -- Second pass: rank candidates and calculate margin
    SELECT
        a.*,
        ROW_NUMBER() OVER (PARTITION BY a.SchemaName, a.TableName ORDER BY a.NetScore DESC, a.SignalCount DESC) AS RankInTable,
        MAX(a.NetScore) OVER (PARTITION BY a.SchemaName, a.TableName) AS MaxScore,
        MIN(a.NetScore) OVER (PARTITION BY a.SchemaName, a.TableName) AS MinScore
    FROM Aggregated a
),
WithRunnerUp AS (
    -- Third pass: get runner-up score for margin calculation
    SELECT
        r.*,
        (SELECT TOP 1 r2.NetScore FROM Ranked r2
         WHERE r2.SchemaName = r.SchemaName AND r2.TableName = r.TableName AND r2.RankInTable = 2) AS RunnerUpScore
    FROM Ranked r
)
INSERT INTO dbo.EvidenceScore(SchemaName, TableName, OwnerCandidate, NetScore, SignalCount, DistinctSignalTypes, Signals, RankInTable, WinningMargin, Confidence)
SELECT
    wr.SchemaName,
    wr.TableName,
    wr.OwnerCandidate,
    wr.NetScore,
    wr.SignalCount,
    wr.DistinctSignalTypes,
    wr.Signals,
    wr.RankInTable,
    -- Winning margin: difference between this candidate and runner-up
    CASE
        WHEN wr.RankInTable = 1 THEN wr.NetScore - ISNULL(wr.RunnerUpScore, wr.MinScore)
        ELSE 0
    END AS WinningMargin,
    -- Simplified confidence: 50 + Margin + TierBonus (capped at 100)
    CASE
        WHEN wr.RankInTable = 1 THEN
            CASE
                WHEN (50 + (wr.NetScore - ISNULL(wr.RunnerUpScore, wr.MinScore)) + (wr.DistinctSignalTypes * 2)) > 100
                THEN 100.00
                ELSE CAST((50 + (wr.NetScore - ISNULL(wr.RunnerUpScore, wr.MinScore)) + (wr.DistinctSignalTypes * 2)) AS DECIMAL(5,2))
            END
        ELSE 50.00
    END AS Confidence
FROM WithRunnerUp wr
WHERE wr.RankInTable = 1; -- Only store the winner per table

-- Populate SourcePolicyPlan
-- FIX (architectural): SourcePolicy answers "which databases contribute rows" -- that is a function of
-- ExistsInHO/ExistsInRegion, NOT of IdentityOwner. IdentityOwner answers "who owns the identity", a
-- different question. The two are correlated but not equivalent (e.g. Composite/Shared tables can exist
-- in both HO and Region and must get SourcePolicy=BOTH, not fall through to TEMPLATE_ONLY).
-- IdentityOwner is retained only to drive MergePriority/MergeOrder tie-breaking during merge.
-- Every IdentityOwner value Layer 2 can currently produce (HO, Region, Composite, Shared, Generated,
-- Unknown) is handled explicitly below with no catch-all business decision. If IdentityPlan ever emits
-- an owner value outside this set, the build THROWs immediately instead of silently mis-routing tables.
IF EXISTS(SELECT 1 FROM dbo.IdentityPlan WHERE IdentityOwner NOT IN(N'HO',N'Region',N'Composite',N'Shared',N'Generated',N'Unknown'))
BEGIN
    DECLARE @BadOwner NVARCHAR(32) = (SELECT TOP 1 IdentityOwner FROM dbo.IdentityPlan WHERE IdentityOwner NOT IN(N'HO',N'Region',N'Composite',N'Shared',N'Generated',N'Unknown'));
    DECLARE @BadOwnerMsg NVARCHAR(400) = N'SourcePolicyPlan: unrecognized IdentityOwner value ['+ISNULL(@BadOwner,N'NULL')+N'] found in IdentityPlan. Add an explicit SourcePolicy rule for this owner before proceeding.';
    THROW 51020, @BadOwnerMsg, 1;
END

INSERT INTO dbo.SourcePolicyPlan(SchemaName, TableName, SourcePolicy, MergePriority, MergeOrder,
    ConflictPolicy, GovernanceStatus, PolicyRule, ReasonCode, ReasonText)
SELECT
    ip.SchemaName,
    ip.TableName,
    -- SourcePolicy: derived from actual existence, not identity ownership
    CASE
        WHEN ISNULL(ip.ExistsInHO,0) = 1 AND ISNULL(ip.ExistsInRegion,0) = 1 THEN N'BOTH'
        WHEN ISNULL(ip.ExistsInHO,0) = 1 AND ISNULL(ip.ExistsInRegion,0) = 0 THEN N'HO_ONLY'
        WHEN ISNULL(ip.ExistsInHO,0) = 0 AND ISNULL(ip.ExistsInRegion,0) = 1 THEN N'REGION_ONLY'
        WHEN ISNULL(ip.ExistsInHO,0) = 0 AND ISNULL(ip.ExistsInRegion,0) = 0 THEN N'TEMPLATE_ONLY'
        -- unreachable given the IdentityOwner guard above plus ExistsInHO/ExistsInRegion being BIT NOT NULL-defaulted upstream,
        -- but fail loud rather than silently defaulting if it ever is
        ELSE N'UNRESOLVED'
    END AS SourcePolicy,
    -- MergePriority/MergeOrder: IdentityOwner's proper job -- sequencing, not sourcing
    CASE ip.IdentityOwner
        WHEN N'HO'        THEN 0
        WHEN N'Composite'  THEN 500
        WHEN N'Shared'     THEN 500
        WHEN N'Region'     THEN 1000
        WHEN N'Generated'  THEN 1500
        WHEN N'Unknown'    THEN 2000
    END AS MergePriority,
    CASE ip.IdentityOwner
        WHEN N'HO'        THEN N'HO_FIRST'
        WHEN N'Composite'  THEN N'HO_FIRST'
        WHEN N'Shared'     THEN N'HO_FIRST'
        WHEN N'Region'     THEN N'REGION_AFTER_HO'
        WHEN N'Generated'  THEN N'ANY'
        WHEN N'Unknown'    THEN N'ANY'
    END AS MergeOrder,

    CASE WHEN ip.IdentityOwner = N'Unknown' THEN N'PENDING_REVIEW' ELSE N'APPROVED' END AS GovernanceStatus,
    CASE

        WHEN ISNULL(ip.ExistsInHO,0) = 1 AND ISNULL(ip.ExistsInRegion,0) = 0 THEN N'SPP_HO_ONLY'

        WHEN ISNULL(ip.ExistsInHO,0) = 0 AND ISNULL(ip.ExistsInRegion,0) = 1 THEN N'Table exists in region(s) only (IdentityOwner='+ip.IdentityOwner+N')'

-- Populate StrategyPlan WITH VALIDATION
INSERT INTO dbo.StrategyPlan(SchemaName, TableName, UniquenessStrategy, MigrationStrategy,
    CompositeScenarioId, PKType, RequiredChildAction, IsUnsupported, UnsupportedReason,
    StrategyRule, ReasonCode, ReasonText, Confidence, RuleId)
SELECT
    ip.SchemaName,
    ip.TableName,
    CASE
        -- FIX: Prevent HO_ONLY + REKEY contradiction
        WHEN spp.SourcePolicy = N'HO_ONLY' THEN N'Preserve'
        WHEN ip.IdentityType = N'Composite' AND td.HasBranchId = 0 THEN N'ExpandPK'
        WHEN ip.IdentityType = N'Numeric' THEN N'Rekey'
        WHEN ip.IdentityType = N'Guid' THEN N'Preserve'
        ELSE N'Preserve'
    END AS UniquenessStrategy,
    CASE
        -- FIX: Prevent HO_ONLY + REKEY contradiction
        WHEN spp.SourcePolicy = N'HO_ONLY' THEN N'Preserve'
        WHEN ip.IdentityType = N'Composite' AND td.HasBranchId = 0 THEN N'ExpandPK'
        WHEN ip.IdentityType = N'Numeric' AND spp.SourcePolicy = N'REGION_ONLY' THEN N'Rekey'
        WHEN ip.IdentityType = N'Guid' THEN N'Preserve'
        ELSE N'Preserve'
    END AS MigrationStrategy,
    0 AS CompositeScenarioId,
    ISNULL(td.PKDataType, N'Unknown') AS PKType,
    CASE
        WHEN ip.IdentityType = N'Numeric' AND spp.SourcePolicy = N'REGION_ONLY' THEN N'RewriteFK'
        WHEN ip.IdentityType = N'Composite' AND td.HasBranchId = 0 THEN N'AddBranchId'
        ELSE N'None'
    END AS RequiredChildAction,
    0 AS IsUnsupported,
    NULL AS UnsupportedReason,
    CASE
        WHEN spp.SourcePolicy = N'HO_ONLY' THEN N'STRAT_HO_PRESERVE'
        WHEN ip.IdentityType = N'Composite' AND td.HasBranchId = 0 THEN N'STRAT_COMP_EXPAND'
        WHEN ip.IdentityType = N'Numeric' AND spp.SourcePolicy = N'REGION_ONLY' THEN N'STRAT_REGION_REKEY'
        WHEN ip.IdentityType = N'Guid' THEN N'STRAT_GUID'
        ELSE N'STRAT_DEFAULT'
    END AS StrategyRule,
    CASE
        WHEN spp.SourcePolicy = N'HO_ONLY' THEN N'STRAT_HO_PRESERVE'
        WHEN ip.IdentityType = N'Composite' AND td.HasBranchId = 0 THEN N'STRAT_COMP_EXPAND'
        WHEN ip.IdentityType = N'Numeric' AND spp.SourcePolicy = N'REGION_ONLY' THEN N'STRAT_REGION_REKEY'
        WHEN ip.IdentityType = N'Guid' THEN N'STRAT_GUID'
        ELSE N'STRAT_DEFAULT'
    END AS ReasonCode,
    CASE
        WHEN spp.SourcePolicy = N'HO_ONLY' THEN N'HO-only table: Preserve PK'
        WHEN ip.IdentityType = N'Composite' AND td.HasBranchId = 0 THEN N'Composite PK without BranchId: ExpandPK'
        WHEN ip.IdentityType = N'Numeric' AND spp.SourcePolicy = N'REGION_ONLY' THEN N'Region numeric PK: Rekey'
        WHEN ip.IdentityType = N'Guid' THEN N'GUID PK: Preserve'
        ELSE N'Default: Preserve'
    END AS ReasonText,
    80 AS Confidence,
    NULL AS RuleId
FROM dbo.IdentityPlan ip
JOIN dbo.SourcePolicyPlan spp ON spp.SchemaName = ip.SchemaName AND spp.TableName = ip.TableName
JOIN dbo.TableDetails td ON td.DatabaseName = @TemplateDB AND td.SchemaName = ip.SchemaName AND td.TableName = ip.TableName;

;WITH ScoredEvidence AS (
SELECT ie.SchemaName, ie.TableName, ec.AppliesToOwner, ec.EvidenceKey, ec.EvidenceGroup, ec.Direction,
ec.BaseWeight, ec.MaxWeight, ec.DecayRate, ie.RawMagnitude, ie.EvidenceValue,
ie.RawMagnitude * ec.Direction *

THEN (SELECT MIN(v) FROM (VALUES (ec.BaseWeight * POWER(ec.DecayRate, ie.RawMagnitude - 1)), (ec.MaxWeight)) x(v))
WHEN ec.DecayRate IS NOT NULL THEN ec.BaseWeight * POWER(ec.DecayRate, ie.RawMagnitude - 1)
WHEN ec.MaxWeight IS NOT NULL THEN (SELECT MIN(v) FROM (VALUES (ec.BaseWeight * ie.RawMagnitude), (ec.MaxWeight)) x(v))
ELSE ec.BaseWeight * ie.RawMagnitude
END AS WeightedScore
FROM dbo.IdentityEvidence ie
JOIN dbo.EvidenceCatalog ec ON ec.EvidenceKey = ie.EvidenceKey
WHERE ec.IsActive = 1
),
OwnerScores AS (
SELECT SchemaName, TableName, AppliesToOwner,
SUM(WeightedScore) AS TotalScore,

COUNT(DISTINCT EvidenceGroup) AS EvidenceBreadth
FROM ScoredEvidence GROUP BY SchemaName, TableName, AppliesToOwner
),

-- FIX: IdentityResolution was locked to Pass 1's default (often 'Rekey') and never revisited when V10
